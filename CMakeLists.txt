cmake_minimum_required(VERSION 3.16)
project(marker_6dof_tracking VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(MARKER6DOF_BUILD_TOOLS "Build existing replay, TCP and rig calibration tools" OFF)
set(DFMARKER_EXPECT_OPENCV_VERSION "" CACHE STRING "Require an exact OpenCV version")
find_package(OpenCV 4.5.4 REQUIRED COMPONENTS core imgproc calib3d)
if(DFMARKER_EXPECT_OPENCV_VERSION AND NOT OpenCV_VERSION VERSION_EQUAL DFMARKER_EXPECT_OPENCV_VERSION)
    message(FATAL_ERROR "Expected OpenCV ${DFMARKER_EXPECT_OPENCV_VERSION}, found ${OpenCV_VERSION}")
endif()
if(OpenCV_VERSION VERSION_LESS 4.7)
    set(marker_detection_module aruco)
else()
    set(marker_detection_module objdetect)
endif()
find_package(OpenCV 4.5.4 REQUIRED COMPONENTS core imgproc calib3d ${marker_detection_module})
set(marker_sdk_opencv_libraries ${OpenCV_LIBS})
find_package(nlohmann_json 3.2 QUIET)
if(NOT nlohmann_json_FOUND)
    find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp DOC "Directory containing nlohmann/json.hpp")
    if(NOT NLOHMANN_JSON_INCLUDE_DIR)
        message(FATAL_ERROR "Install nlohmann_json or set NLOHMANN_JSON_INCLUDE_DIR")
    endif()
    add_library(marker_json INTERFACE)
    target_include_directories(marker_json INTERFACE "${NLOHMANN_JSON_INCLUDE_DIR}")
    add_library(nlohmann_json::nlohmann_json ALIAS marker_json)
endif()
option(MARKER6DOF_ENABLE_REALSENSE "Build the live D455 input when the C++ SDK is available" ON)
set(MARKER6DOF_WITH_REALSENSE 0)
if(MARKER6DOF_BUILD_TOOLS AND MARKER6DOF_ENABLE_REALSENSE)
    find_package(realsense2 CONFIG QUIET)
    if(realsense2_FOUND)
        set(MARKER6DOF_WITH_REALSENSE 1)
    endif()
endif()

add_library(marker6dof STATIC cpp/core.cpp cpp/camera.cpp)
set_target_properties(marker6dof PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(marker6dof PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(marker6dof PUBLIC ${marker_sdk_opencv_libraries} nlohmann_json::nlohmann_json)
target_compile_definitions(marker6dof PUBLIC MARKER6DOF_WITH_REALSENSE=${MARKER6DOF_WITH_REALSENSE})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(marker6dof PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(dfmarker SHARED cpp/dfmarker_api.cpp)
add_library(dfmarker::dfmarker ALIAS dfmarker)
set_target_properties(dfmarker PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR}
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_definitions(dfmarker PRIVATE DFMARKER_BUILD)
target_include_directories(dfmarker PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
target_link_libraries(dfmarker PRIVATE marker6dof)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dfmarker PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(dfmarker PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/cmake/dfmarker.exports")
    set_property(TARGET dfmarker APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/dfmarker.exports")
endif()

if(MARKER6DOF_BUILD_TOOLS)
    find_package(OpenCV 4.5.4 REQUIRED COMPONENTS videoio)
    add_executable(marker_tracker cpp/main.cpp)
    target_link_libraries(marker_tracker PRIVATE marker6dof opencv_videoio)
    add_executable(marker_calibrate_rig cpp/calibrate_rig.cpp)
    target_link_libraries(marker_calibrate_rig PRIVATE marker6dof)
    install(TARGETS marker_tracker marker_calibrate_rig RUNTIME DESTINATION bin)
endif()
if(MARKER6DOF_BUILD_TOOLS AND MARKER6DOF_WITH_REALSENSE)
    if(TARGET realsense2::realsense2)
        target_link_libraries(marker_tracker PRIVATE realsense2::realsense2)
    elseif(TARGET realsense2)
        target_link_libraries(marker_tracker PRIVATE realsense2)
    else()
        target_link_libraries(marker_tracker PRIVATE ${realsense2_LIBRARY})
        target_include_directories(marker_tracker PRIVATE ${realsense2_INCLUDE_DIR})
    endif()
endif()
message(STATUS "dfmarker SDK: OpenCV ${OpenCV_VERSION}; tools ${MARKER6DOF_BUILD_TOOLS}; live D455 ${MARKER6DOF_WITH_REALSENSE}")

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS dfmarker EXPORT dfmarkerTargets
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/dfmarker_api.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY configs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/dfmarker/configs
    PATTERN "*_local.json" EXCLUDE PATTERN "*_real.json" EXCLUDE)
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/dfmarker/examples)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/dfmarker)
configure_package_config_file(cmake/dfmarkerConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/dfmarkerConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_DATADIR}/dfmarker/cmake)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/dfmarkerConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
configure_file(cmake/build-info.txt.in ${CMAKE_CURRENT_BINARY_DIR}/build-info.txt @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/build-info.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/dfmarker)
install(EXPORT dfmarkerTargets NAMESPACE dfmarker:: DESTINATION ${CMAKE_INSTALL_DATADIR}/dfmarker/cmake)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/dfmarkerConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/dfmarkerConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_DATADIR}/dfmarker/cmake)

include(CTest)
if(BUILD_TESTING)
    add_executable(marker_core_test tests/core_test.cpp)
    target_link_libraries(marker_core_test PRIVATE marker6dof)
    target_compile_definitions(marker_core_test PRIVATE MARKER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME marker_core_test COMMAND marker_core_test)
    set_tests_properties(marker_core_test PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(marker_api_test tests/api_test.cpp)
    target_link_libraries(marker_api_test PRIVATE dfmarker marker6dof)
    add_test(NAME marker_api_test COMMAND marker_api_test)
    add_executable(marker_c_api_test examples/c_api/main.c)
    set_target_properties(marker_c_api_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
    target_link_libraries(marker_c_api_test PRIVATE dfmarker)
    add_test(NAME marker_c_api_test COMMAND marker_c_api_test "${CMAKE_CURRENT_SOURCE_DIR}/configs/sdk_demo.json")
endif()
